## Key Functions

### createSchedule
Creates or reuses a schedule for a given organization and month.

```typescript
export async function createSchedule(orgId: string, month: string) {
  // 1. Check for existing schedule
  const { data: existing } = await supabase
    .from('schedules')
    .select('*')
    .eq('organization_id', orgId)
    .eq('month', month)
    .maybeSingle();

  // 2. Reuse existing schedule (reset status)
  if (existing) {
    const { data, error } = await supabase
      .from('schedules')
      .update({
        status: 'created',
        hard_score: null,
        soft_score: null,
        solver_execution_id: null,
      })
      .eq('id', existing.id)
      .select()
      .single();

    if (error) throw new Error(`근무표 생성 실패: ${error.message}`);
    return data as Schedule;
  }

  // 3. Create new schedule
  const { data, error } = await supabase
    .from('schedules')
    .insert({
      organization_id: orgId,
      month,
      status: 'created',
    })
    .select()
    .single();

  if (error) throw new Error(`근무표 생성 실패: ${error.message}`);
  return data as Schedule;
}
```

### getScheduleAssignments
Paginated query for schedule assignments.

```typescript
export async function getScheduleAssignments(
  scheduleId: string
): Promise<AssignmentMap> {
  // Get all assignments without pagination for schedule grid
  const { data, error } = await supabase
    .from('schedule_assignments')
    .select('*')
    .eq('schedule_id', scheduleId);

  if (error) {
    throw new Error(`근무표 배정 조회 실패: ${error.message}`);
  }

  // Transform to AssignmentMap
  const map: AssignmentMap = {};
  for (const assignment of data || []) {
    if (!map[assignment.employee_id]) {
      map[assignment.employee_id] = {};
    }
    map[assignment.employee_id][assignment.date] = assignment.shifts || null;
  }

  return map;
}
```

### completeSchedule
Updates schedule status with solver results.

```typescript
export async function completeSchedule(
  scheduleId: string,
  result: SolverResult
): Promise<void> {
  const { error } = await supabase
    .from('schedules')
    .update({
      status: 'complete',
      hard_score: result.hardScore,
      soft_score: result.softScore,
      updated_at: new Date().toISOString(),
    })
    .eq('id', scheduleId);

  if (error) {
    throw new Error(`근무표 완료 처리 실패: ${error.message}`);
  }
}
```
