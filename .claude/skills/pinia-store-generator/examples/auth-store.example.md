# Auth Store Example

## Real Example from Codebase

**File**: [src/stores/auth.ts](../src/stores/auth.ts)

## Overview
Simple authentication store with Supabase session management.

## Key Features
- User and session state
- Authentication status getter
- Login/logout actions
- Session persistence with Supabase

## Usage Pattern

```typescript
import { useAuthStore } from '@/stores/auth';

const authStore = useAuthStore();

// Set user after login
authStore.setUser({
  id: 'user123',
  email: 'user@example.com',
  name: 'John Doe',
});

// Check authentication
if (authStore.isAuthenticated) {
  // User is logged in
}

// Logout
authStore.logout();
```

## Implementation Details

### State Structure
```typescript
const user = ref<User | null>(null);
const session = ref<Session | null>(null);
```

### Getters
```typescript
const isAuthenticated = computed(() => !!user.value);
```

### Actions
```typescript
function setUser(userData: User | null) {
  user.value = userData;
}

function setSession(sessionData: Session | null) {
  session.value = sessionData;
}

function logout() {
  user.value = null;
  session.value = null;
}
```

## Integration with Supabase

This store integrates with Supabase auth:

```typescript
import { supabase } from '@/api/supabase';

// Set up auth state listener
supabase.auth.onAuthStateChange((event, session) => {
  if (event === 'SIGNED_IN') {
    authStore.setUser(session.user);
    authStore.setSession(session);
  } else if (event === 'SIGNED_OUT') {
    authStore.logout();
  }
});
```
